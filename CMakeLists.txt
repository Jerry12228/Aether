cmake_minimum_required(VERSION 3.31)
project(Aether LANGUAGES C CXX)
option(AETHER_CORE_ONLY "Only build the shared core from a Flutter plugin" OFF)
option(AETHER_ENABLE_TEST_HOOKS "Enable explicit local fault test commands" OFF)
get_filename_component(AETHER_ROOT "${CMAKE_CURRENT_LIST_DIR}" ABSOLUTE)
find_program(AETHER_NODE NAMES node.exe node REQUIRED)
execute_process(COMMAND "${AETHER_NODE}" "${AETHER_ROOT}/scripts/version.cjs" --write
  RESULT_VARIABLE VERSION_RESULT OUTPUT_QUIET)
if(NOT VERSION_RESULT EQUAL 0)
  message(FATAL_ERROR "Version generation failed")
endif()
add_library(aether_core SHARED native/core/src/core.cpp native/platform/platform.cpp)
target_compile_features(aether_core PRIVATE cxx_std_20)
target_compile_definitions(aether_core PRIVATE AETHER_CORE_EXPORTS)
target_include_directories(aether_core PUBLIC "${AETHER_ROOT}/native/core/include" "${AETHER_ROOT}/native/platform/include" PRIVATE "${AETHER_ROOT}/build/native/generated")
if(MSVC)
  target_compile_options(aether_core PRIVATE /W4 /WX /EHsc)
endif()
if(NOT AETHER_CORE_ONLY)
  add_library(aether_virtual_display STATIC native/platform/windows/virtual_display_provider.cpp native/platform/windows/virtual_display_mutation.cpp)
  target_compile_features(aether_virtual_display PRIVATE cxx_std_20)
  target_compile_definitions(aether_virtual_display PRIVATE NOMINMAX)
  target_include_directories(aether_virtual_display PUBLIC "${AETHER_ROOT}/native/platform/include")
  target_include_directories(aether_virtual_display PRIVATE "${AETHER_ROOT}/apps/helios")
  target_link_libraries(aether_virtual_display PRIVATE user32 setupapi cfgmgr32 advapi32 bcrypt ole32)
  add_executable(helios apps/helios/main.cpp apps/helios/display_commands.cpp apps/helios/display_registry.cpp apps/helios/display_test_pattern.cpp)
  target_compile_features(helios PRIVATE cxx_std_20)
  target_compile_definitions(helios PRIVATE NOMINMAX)
  target_link_libraries(helios PRIVATE aether_core aether_virtual_display shell32 gdi32)
  # Fault injection belongs exclusively to aether_helios_test, regardless of
  # the legacy cache option. Product builds never accept test-root/provider.
  include(CTest)
  if(BUILD_TESTING)
    add_executable(aether_virtual_display_contract tests/native/virtual_display_contract.cpp)
    target_compile_features(aether_virtual_display_contract PRIVATE cxx_std_20)
    target_link_libraries(aether_virtual_display_contract PRIVATE aether_virtual_display)
    add_test(NAME virtual_display_probe COMMAND aether_virtual_display_contract $<TARGET_FILE:helios>)
    set_tests_properties(virtual_display_probe PROPERTIES TIMEOUT 10)
    add_executable(aether_presentation_contract tests/native/presentation_contract.cpp native/platform/windows/presentation.cpp)
    target_compile_features(aether_presentation_contract PRIVATE cxx_std_20)
    target_include_directories(aether_presentation_contract PRIVATE "${AETHER_ROOT}/native/platform/include")
    target_link_libraries(aether_presentation_contract PRIVATE d3d11 dxgi d2d1 dwrite d3dcompiler windowscodecs ole32)
    add_executable(aether_helios_test apps/helios/main.cpp apps/helios/display_commands.cpp apps/helios/display_registry.cpp apps/helios/display_test_pattern.cpp)
    target_compile_features(aether_helios_test PRIVATE cxx_std_20)
    target_compile_definitions(aether_helios_test PRIVATE AETHER_ENABLE_TEST_HOOKS)
    target_compile_definitions(aether_helios_test PRIVATE NOMINMAX)
    target_link_libraries(aether_helios_test PRIVATE aether_core aether_virtual_display shell32 gdi32)
    foreach(SUITE tracer registry topology failure recovery pattern security)
      add_test(NAME virtual_display_${SUITE} COMMAND aether_virtual_display_contract $<TARGET_FILE:aether_helios_test> ${SUITE} "${AETHER_ROOT}/tests/virtual-display-contract.test.cjs" "${AETHER_NODE}")
      set_tests_properties(virtual_display_${SUITE} PROPERTIES TIMEOUT 30)
    endforeach()
    add_executable(aether_helios_signal tests/native/helios_signal.cpp)
    target_compile_features(aether_helios_signal PRIVATE cxx_std_20)
    add_executable(aether_core_contract tests/native/core_contract.cpp tests/native/c_consumer.c native/platform/platform.cpp)
    target_compile_features(aether_core_contract PRIVATE cxx_std_20)
    set_property(TARGET aether_core_contract PROPERTY C_STANDARD 11)
    target_link_libraries(aether_core_contract PRIVATE aether_core)
    foreach(SUITE tracer lifecycle adapters)
      add_test(NAME core_${SUITE} COMMAND aether_core_contract ${SUITE})
      set_tests_properties(core_${SUITE} PROPERTIES TIMEOUT 60)
    endforeach()
  endif()
endif()
