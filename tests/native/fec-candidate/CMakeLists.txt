cmake_minimum_required(VERSION 3.31)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
project(AetherFecResearch LANGUAGES CXX)
get_filename_component(AETHER_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(RESEARCH "${AETHER_ROOT}/build/phase06/fec-alternatives")
file(READ "${RESEARCH}/botan-build/build/build_config.json" CONFIG)
string(JSON COUNT LENGTH "${CONFIG}" lib_build_info)
math(EXPR LAST "${COUNT}-1")
set(SOURCES)
foreach(I RANGE 0 ${LAST})
  string(JSON SOURCE GET "${CONFIG}" lib_build_info ${I} src)
  list(APPEND SOURCES "${RESEARCH}/botan/${SOURCE}")
endforeach()
add_library(fec_research STATIC ${SOURCES})
target_include_directories(fec_research PUBLIC "${RESEARCH}/botan-build/build/include/public")
target_include_directories(fec_research PRIVATE "${RESEARCH}/botan-build/build/include/internal")
target_compile_features(fec_research PUBLIC cxx_std_20)
target_compile_definitions(fec_research PRIVATE NOMINMAX)
add_executable(fec_candidate candidate.cpp)
target_link_libraries(fec_candidate PRIVATE fec_research)
if(MSVC)
  target_compile_options(fec_research PRIVATE /W4 /WX /Zc:preprocessor /bigobj)
  target_compile_options(fec_candidate PRIVATE /W4 /WX)
endif()
option(AETHER_FEC_SANITIZE "Instrument every research library source and caller with AddressSanitizer" OFF)
if(AETHER_FEC_SANITIZE)
  if(NOT MSVC)
    message(FATAL_ERROR "This evidence target is for the current Windows MSVC executor only")
  endif()
  target_compile_options(fec_research PRIVATE /fsanitize=address /Zi)
  target_compile_options(fec_candidate PRIVATE /fsanitize=address /Zi)
  target_link_options(fec_candidate PRIVATE /DEBUG /INCREMENTAL:NO)
endif()
