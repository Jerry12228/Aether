cmake_minimum_required(VERSION 3.31)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
project(AetherProtocol LANGUAGES C CXX)
get_filename_component(AETHER_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(AETHER_DEPS "${AETHER_ROOT}/build/phase06/dependencies" CACHE PATH "Audited isolated source builds")
if(NOT EXISTS "${AETHER_DEPS}/audit.json")
  message(FATAL_ERROR "Run the protocol harness --prepare and --build-openssl first")
endif()
file(READ "${AETHER_DEPS}/audit.json" AUDIT)
string(JSON ARCHIVE_SHA GET "${AUDIT}" dependencies 0 archiveSha256)
if(NOT ARCHIVE_SHA STREQUAL "603f5602e2eef00d77fbd429d34dcd5822bb301757a1bc9cdb24c670f1eb859a")
  message(FATAL_ERROR "OpenSSL source provenance mismatch")
endif()
foreach(INDEX 0 1)
  string(JSON NAME GET "${AUDIT}" dependencies ${INDEX} name)
  string(JSON SHA GET "${AUDIT}" dependencies ${INDEX} sourceSha)
  string(JSON EXPECTED GET "${AUDIT}" dependencies ${INDEX} archiveSha256)
  file(SHA256 "${AETHER_DEPS}/${NAME}-${SHA}.tar.gz" ACTUAL)
  if(NOT ACTUAL STREQUAL EXPECTED)
    message(FATAL_ERROR "Dependency archive drift")
  endif()
  string(JSON NOTICE_COUNT LENGTH "${AUDIT}" dependencies ${INDEX} notices)
  math(EXPR LAST_NOTICE "${NOTICE_COUNT}-1")
  foreach(N RANGE 0 ${LAST_NOTICE})
    string(JSON NOTICE_PATH GET "${AUDIT}" dependencies ${INDEX} notices ${N} path)
    string(JSON NOTICE_SHA GET "${AUDIT}" dependencies ${INDEX} notices ${N} sha256)
    file(SHA256 "${AETHER_DEPS}/${NAME}/${NOTICE_PATH}" ACTUAL)
    if(NOT ACTUAL STREQUAL NOTICE_SHA)
      message(FATAL_ERROR "Dependency notice drift")
    endif()
  endforeach()
endforeach()
string(JSON SRTP_SHA GET "${AUDIT}" dependencies 1 sourceSha)
string(JSON SRTP_ARCHIVE GET "${AUDIT}" dependencies 1 archiveSha256)
if(NOT SRTP_SHA STREQUAL "24b3bf8f19b6f5ab4cd2bcceb4f4064efca86fd5" OR
   NOT SRTP_ARCHIVE STREQUAL "031bf6c0b38c02c3425bf8b4a24386f953ec4ab3df203adb6c22234d2a63f1ba")
  message(FATAL_ERROR "SRTP source provenance mismatch")
endif()
set(OPENSSL_ROOT_DIR "${AETHER_DEPS}/openssl-install")
set(OPENSSL_USE_STATIC_LIBS TRUE)
set(OPENSSL_INCLUDE_DIR "${OPENSSL_ROOT_DIR}/include" CACHE PATH "Pinned headers" FORCE)
set(OPENSSL_SSL_LIBRARY "${OPENSSL_ROOT_DIR}/lib/libssl.lib" CACHE FILEPATH "Pinned SSL" FORCE)
set(OPENSSL_CRYPTO_LIBRARY "${OPENSSL_ROOT_DIR}/lib/libcrypto.lib" CACHE FILEPATH "Pinned crypto" FORCE)
find_package(OpenSSL 3.5.9 EXACT REQUIRED)
set(ENABLE_OPENSSL ON CACHE BOOL "SRTP AEAD backend" FORCE)
set(LIBSRTP_TEST_APPS ON CACHE BOOL "Independent upstream KATs" FORCE)
set(ENABLE_WARNINGS_AS_ERRORS ON CACHE BOOL "Dependency warnings are errors" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static dependency" FORCE)
add_subdirectory("${AETHER_DEPS}/libsrtp" "${CMAKE_CURRENT_BINARY_DIR}/libsrtp")
file(READ "${AETHER_ROOT}/docs/phase06/FEC-REUSE-DECISION.json" FEC_DECISION)
string(JSON FEC_CHOICE GET "${FEC_DECISION}" choice)
if(NOT FEC_CHOICE STREQUAL "approve-botan-prototype-reuse")
  message(FATAL_ERROR "Concrete FEC reuse approval is required")
endif()
foreach(I RANGE 0 1)
  string(JSON MATERIAL GET "${FEC_DECISION}" reviewedMaterials ${I} path)
  string(JSON EXPECTED GET "${FEC_DECISION}" reviewedMaterials ${I} sha256)
  file(SHA256 "${AETHER_ROOT}/${MATERIAL}" ACTUAL)
  if(NOT ACTUAL STREQUAL EXPECTED)
    message(FATAL_ERROR "Approved FEC material drift")
  endif()
endforeach()
file(READ "${AETHER_ROOT}/docs/phase06/FEC-ALTERNATIVES.json" FEC_LEDGER)
string(JSON FEC_PIN GET "${FEC_LEDGER}" selected sourceSha)
string(JSON FEC_ARCHIVE GET "${FEC_LEDGER}" selected archiveSha256)
if(NOT FEC_PIN STREQUAL "2a81eef56c96c237e590c27f9a75e60317c9c700" OR
   NOT FEC_ARCHIVE STREQUAL "714d6c52c6d752808e8784df21c146e176e91c57c99ee72c08eff4439fcff872")
  message(FATAL_ERROR "Unapproved FEC pin")
endif()
set(FEC_ROOT "${AETHER_ROOT}/build/phase06/fec-alternatives")
file(SHA256 "${FEC_ROOT}/botan-${FEC_PIN}.tar.gz" ACTUAL)
if(NOT ACTUAL STREQUAL FEC_ARCHIVE)
  message(FATAL_ERROR "FEC archive drift")
endif()
string(JSON FEC_COUNT LENGTH "${FEC_LEDGER}" selected sourceFiles)
math(EXPR FEC_LAST "${FEC_COUNT}-1")
set(FEC_SOURCES)
foreach(I RANGE 0 ${FEC_LAST})
  string(JSON INPUT GET "${FEC_LEDGER}" selected sourceFiles ${I} path)
  string(JSON EXPECTED GET "${FEC_LEDGER}" selected sourceFiles ${I} sha256)
  file(SHA256 "${AETHER_ROOT}/${INPUT}" ACTUAL)
  if(NOT ACTUAL STREQUAL EXPECTED)
    message(FATAL_ERROR "Approved FEC source/header/notice drift: ${INPUT}")
  endif()
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${AETHER_ROOT}/${INPUT}")
  if(INPUT MATCHES "\\.cpp$")
    list(APPEND FEC_SOURCES "${AETHER_ROOT}/${INPUT}")
  endif()
endforeach()
list(LENGTH FEC_SOURCES FEC_CPP_COUNT)
if(NOT FEC_CPP_COUNT EQUAL 23)
  message(FATAL_ERROR "Approved scalar FEC source set mismatch")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  "${AETHER_ROOT}/docs/phase06/FEC-REUSE-DECISION.json"
  "${AETHER_ROOT}/docs/phase06/FEC-ALTERNATIVES.md"
  "${AETHER_ROOT}/docs/phase06/FEC-ALTERNATIVES.json")
add_library(aether_fec STATIC ${FEC_SOURCES})
target_compile_features(aether_fec PUBLIC cxx_std_20)
target_include_directories(aether_fec PUBLIC "${FEC_ROOT}/botan-build/build/include/public")
target_include_directories(aether_fec PRIVATE "${FEC_ROOT}/botan-build/build/include/internal")
target_compile_definitions(aether_fec PRIVATE NOMINMAX)
if(MSVC)
  target_compile_options(aether_fec PRIVATE /W4 /WX /Zc:preprocessor /bigobj)
endif()
add_library(aether_protocol SHARED src/protocol.cpp src/control.cpp src/admission.cpp src/secure_transport.cpp src/media_packetization.cpp)
target_compile_features(aether_protocol PRIVATE cxx_std_20)
target_compile_definitions(aether_protocol PRIVATE AETHER_PROTOCOL_EXPORTS NOMINMAX)
target_include_directories(aether_protocol PUBLIC include)
target_link_libraries(aether_protocol PRIVATE OpenSSL::SSL libSRTP::srtp2 aether_fec)
if(WIN32)
  target_link_libraries(aether_protocol PRIVATE ws2_32 crypt32)
endif()
if(MSVC)
  target_compile_options(aether_protocol PRIVATE /W4 /WX /EHsc)
endif()
find_program(AETHER_NODE NAMES node.exe node REQUIRED)
include(CTest)
if(BUILD_TESTING)
  add_executable(aether_protocol_control_client "${AETHER_ROOT}/tests/native/protocol_control.cpp")
  target_link_libraries(aether_protocol_control_client PRIVATE aether_protocol)
  target_compile_features(aether_protocol_control_client PRIVATE cxx_std_20)
  target_compile_definitions(aether_protocol_control_client PRIVATE NOMINMAX)
  add_library(aether_protocol_control_contract STATIC src/protocol.cpp src/control.cpp src/admission.cpp src/secure_transport.cpp src/media_packetization.cpp)
  target_compile_definitions(aether_protocol_control_contract PRIVATE AETHER_PROTOCOL_EXPORTS AETHER_CONTROL_FIXTURE NOMINMAX)
  target_include_directories(aether_protocol_control_contract PUBLIC include src)
  target_compile_features(aether_protocol_control_contract PRIVATE cxx_std_20)
  target_link_libraries(aether_protocol_control_contract PRIVATE OpenSSL::SSL libSRTP::srtp2 aether_fec ws2_32 crypt32)
  add_executable(aether_protocol_control_host "${AETHER_ROOT}/tests/native/protocol_control.cpp")
  target_compile_definitions(aether_protocol_control_host PRIVATE AETHER_CONTROL_HOST AETHER_CONTROL_FIXTURE AETHER_PROTOCOL_EXPORTS NOMINMAX)
  target_link_libraries(aether_protocol_control_host PRIVATE aether_protocol_control_contract OpenSSL::SSL libSRTP::srtp2)
  target_compile_features(aether_protocol_control_host PRIVATE cxx_std_20)
  add_dependencies(aether_protocol_control_client aether_protocol_control_host)
  add_test(NAME protocol_control COMMAND aether_protocol_control_client --orchestrate)
  set_tests_properties(protocol_control PROPERTIES LABELS protocol-fast TIMEOUT 30)
  add_executable(aether_media_security "${AETHER_ROOT}/tests/native/protocol_media_security.cpp" src/secure_transport.cpp src/media_packetization.cpp)
  target_include_directories(aether_media_security PRIVATE include src)
  target_compile_definitions(aether_media_security PRIVATE NOMINMAX)
  target_compile_features(aether_media_security PRIVATE cxx_std_20)
  target_link_libraries(aether_media_security PRIVATE OpenSSL::SSL libSRTP::srtp2 aether_fec)
  if(WIN32)
    target_link_libraries(aether_media_security PRIVATE ws2_32 crypt32)
  endif()
  if(MSVC)
    target_compile_options(aether_media_security PRIVATE /W4 /WX)
  endif()
  add_executable(aether_protocol_tracer tools/tracer.cpp)
  add_dependencies(aether_protocol_tracer aether_media_security)
  set_property(TARGET aether_protocol_tracer APPEND PROPERTY LINK_DEPENDS
    "${AETHER_ROOT}/tests/protocol-tracer.test.cjs" "${AETHER_ROOT}/tests/protocol-multistream.test.cjs"
    "${AETHER_ROOT}/tests/native/protocol_media_security.cpp")
  target_compile_features(aether_protocol_tracer PRIVATE cxx_std_20)
  target_link_libraries(aether_protocol_tracer PRIVATE aether_protocol OpenSSL::Crypto)
  add_custom_command(TARGET aether_protocol_tracer POST_BUILD
    COMMAND "${AETHER_NODE}" "${AETHER_ROOT}/tests/protocol-tracer.test.cjs"
      --manifest "$<TARGET_FILE:aether_protocol_tracer>" "$<CONFIG>" "${CMAKE_CURRENT_BINARY_DIR}")
  add_test(NAME protocol_tracer COMMAND "${AETHER_NODE}" --test "${AETHER_ROOT}/tests/protocol-tracer.test.cjs")
  set_tests_properties(protocol_tracer PROPERTIES LABELS protocol-fast TIMEOUT 30)
  add_test(NAME protocol_multistream COMMAND "${AETHER_NODE}" --test "${AETHER_ROOT}/tests/protocol-multistream.test.cjs")
  add_test(NAME protocol_recovery COMMAND "${AETHER_NODE}" --test "${AETHER_ROOT}/tests/protocol-recovery.test.cjs")
  set_tests_properties(protocol_recovery PROPERTIES LABELS protocol-fast TIMEOUT 90)
  add_custom_target(protocol_fast COMMAND "${CMAKE_CTEST_COMMAND}" -C "$<CONFIG>" -L protocol-fast --output-on-failure --no-tests=error DEPENDS aether_protocol_tracer aether_protocol_control_client)
  set_tests_properties(protocol_multistream PROPERTIES LABELS protocol-fast TIMEOUT 30)
endif()
